/* AI 文档：综合文本分析 */
AI.analyze=function(text){
  text=text||'';
  const plain=text.replace(/\s/g,'');
  const sents=splitSentences(text);
  const kws=keywordFreq(text,15);
  const summary=extractSummary(text,5).join('');
  const outline=buildOutline(text);
  const points=extractSummary(text,8).map((s,i)=>`${i+1}. ${s}`);
  // 可读性统计
  const avgLen=sents.length?Math.round(plain.length/sents.length):0;
  return {
    chars:plain.length, sents:sents.length, paras:text.split(/\n+/).filter(x=>x.trim()).length, avgLen,
    keywords:kws, summary, points, outline,
    read:avgLen>45?'句子偏长，建议适当断句以提升可读性':avgLen<12&&sents.length>3?'句子偏短、略显零散，可适当合并':'语句长度适中，整体可读性良好'
  };
};
/* 文档问答：基于文档原文检索后组织回答 */
AI.docAsk=function(query, rawText, agent){
  const sents=splitSentences(rawText||'');
  if(!sents.length)return '该文档暂未解析出可用文本（可能是扫描件或无文本层），请更换电子版 PDF，或在文档管理中重新上传。';
  const qk=keywordFreq(query,6).concat(query.replace(/[的了吗呢？?\s]/g,'').split('').filter(x=>x.length));
  const scored=sents.map((s,idx)=>{let sc=0;for(const w of qk){if(s.includes(w))sc+=w.length>=2?2:0.4;}return {s,idx,sc};}).filter(x=>x.sc>0).sort((a,b)=>b.sc-a.sc||a.idx-b.idx);
  const ev=scored.slice(0,4).sort((a,b)=>a.idx-b.idx).map(x=>x.s);
  if(!ev.length){
    // 无直接命中：给文档概要
    return `文档中未检索到与“${query}”直接对应的表述。根据文档整体内容，可为您提供以下概要：\n${extractSummary(rawText,4).join('')}\n您可以换个关键词再问，例如直接引用文档中的专有名词。`;
  }
  const ans=ev.join('');
  return `根据文档内容，${/[？?]\s*$/.test(query)?'':''}回答如下：\n${ans}\n\n【依据原文】\n${ev.map((s,i)=>`[${i+1}] ${s.length>120?s.slice(0,120)+'…':s}`).join('\n')}`;
};
