let toastTimer={};
function toast(msg, type='success', dur=2400){
  const wrap=$('#toastWrap');
  const icon = type==='success'?ICON.check:type==='warn'?ICON.warn:type==='error'?ICON.error:ICON.info;
  const t=h('div',{class:'toast '+type},[h('span',{html:icon}), h('span',{text:msg})]);
  wrap.appendChild(t);
  setTimeout(()=>{t.style.transition='opacity .3s,transform .3s';t.style.opacity='0';t.style.transform='translateY(-8px)';setTimeout(()=>t.remove(),300);},dur);
}

/* 通用模态：content 为 DOM；返回根节点，close() 关闭 */
let activeModal=null;
function openModal({width=560, title, body, footer, onClose, customClass=''}){
  closeModal();
  const root=$('#modalRoot');
  const closeBtn=h('button',{class:'modal-close',html:ICON.close,onClick:()=>closeModal()});
  const modal=h('div',{class:'modal '+customClass,style:{width:width+'px',maxWidth:'94vw'}}},[
    title?h('div',{class:'modal-head'},[h('h3',{text:title}),closeBtn]):null,
    h('div',{class:'modal-body'},Array.isArray(body)?body:[body]),
    footer?h('div',{class:'modal-foot'},footer):null
  ]);
  const mask=h('div',{class:'modal-mask',onClick:e=>{if(e.target===e.currentTarget)closeModal();}},[modal]);
  root.appendChild(mask);
  activeModal={mask,onClose};
  return {modal,mask,close:closeModal};
}
function closeModal(){ if(activeModal){const{mask,onClose}=activeModal;mask.remove();activeModal=null;onClose&&onClose();} }
