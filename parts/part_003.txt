function streamText(fullText, onChunk, opts={}){
  const speed = opts.speed||14; // 每片毫秒
  // 动态步进：总片数封顶约 100 片，长文也能在数秒内完成、短文保持逐字感
  const step = opts.step||Math.max(2,Math.ceil(fullText.length/100));
  let i=0, stopped=false, timer=null;
  return {
    start(){ return new Promise(resolve=>{
      const tick=()=>{ if(stopped){resolve(i);return;}
        i=Math.min(fullText.length, i+step);
        onChunk(fullText.slice(0,i), i>=fullText.length);
        if (i>=fullText.length){resolve(i);return;}
        timer=setTimeout(tick, speed);
      }; tick();
    });},
    stop(){ stopped=true; if(timer)clearTimeout(timer); }
  };
}
