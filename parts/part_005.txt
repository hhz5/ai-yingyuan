/* 剪贴板：优先 Clipboard API，失败回退 execCommand（兼容 file:// 双击打开） */
function clipboardCopy(text){
  return new Promise((resolve,reject)=>{
    if(navigator.clipboard&&window.isSecureContext!==false&&location.protocol!=='file:'){
      navigator.clipboard.writeText(text).then(resolve).catch(()=>fallback());
    }else fallback();
    function fallback(){
      try{
        const ta=document.createElement('textarea');
        ta.value=text;ta.style.position='fixed';ta.style.top='-9999px';ta.style.opacity='0';
        document.body.appendChild(ta);ta.focus();ta.select();
        const ok=document.execCommand('copy');ta.remove();
        ok?resolve():reject(new Error('copy failed'));
      }catch(e){reject(e);}
    }
  });
}

/* ============================================================
   纯前端 PDF 文本提取（无第三方库）
   支持 FlateDecode(zlib) 解压、ToUnicode CMap(bfchar/bfrange) 还原中文、
   Tj/TJ 文本操作符与排版换行；扫描件无文本层时返回空串由上层判定。
   ============================================================ */
const PdfExtract=(function(){
  async function inflate(bytes){
    try{
      const ds=new DecompressionStream('deflate');
      const stream=new Blob([bytes]).stream().pipeThrough(ds);
      const ab=await new Response(stream).arrayBuffer();
      return new Uint8Array(ab);
    }catch(e){return null;}
  }
  // 字节→二进制字符串（真正逐字节直映；勿用 TextDecoder latin1，其按 windows-1252 映射会破坏 0x80-0x9F）
  function bytesToStr(bytes){let s='';const CH=0x8000;for(let i=0;i<bytes.length;i+=CH)s+=String.fromCharCode.apply(null,bytes.subarray(i,i+CH));return s;}
  // 提取全部 stream（优先 /Length，回退正则）
  function* rawStreams(bytes){
    const s=bytesToStr(bytes);
    const re=/(?<!end)stream(\r\n|\n|\r)/g;let m;
    while((m=re.exec(s))){
      const start=m.index+m[0].length;
      // 用最近的 /Length
      const head=s.slice(Math.max(0,m.index-400),m.index);
      const allLen=head.match(/\/Length\s+(\d+)/g);
      const lm=allLen&&allLen.length?/(\d+)/.exec(allLen[allLen.length-1]):null;
      let end;
      if(lm){const len=+lm[1];end=start+len;}
      else{const ei=s.indexOf('endstream',start);end=ei;}
      const chunk=s.slice(start,end).replace(/(\r\n|\n|\r)$/,'');
      yield {text:chunk,head:head};
    }
  }
  function latinBytes(str){const u=new Uint8Array(str.length);for(let i=0;i<str.length;i++)u[i]=str.charCodeAt(i)&0xff;return u;}
  // hex(UTF16-BE) -> 字符串
  function hexToUnicode(hex){
    hex=hex.replace(/\s/g,'');let out='';
    for(let i=0;i+4<=hex.length;i+=4){const code=parseInt(hex.slice(i,i+4),16);if(!isNaN(code))out+=String.fromCharCode(code);}
    return out;
  }
  // 解析全部 ToUnicode CMap，合并为 cid(整数)->字符
  function parseCMap(text,map){
    const blocks=text.match(/beginbfchar[\s\S]*?endbfchar/g)||[];
    blocks.forEach(b=>{
      const re=/<([0-9A-Fa-f]+)>\s*<([0-9A-Fa-f\s]+)>/g;let m;
      while((m=re.exec(b)))map.set(parseInt(m[1],16),hexToUnicode(m[2]));
    });
    const ranges=text.match(/beginbfrange[\s\S]*?endbfrange/g)||[];
    ranges.forEach(b=>{
      const lines=b.split('\n');
      lines.forEach(line=>{
        let m=/<([0-9A-Fa-f]+)>\s*<([0-9A-Fa-f]+)>\s*(?:<([0-9A-Fa-f\s]+)>|\[([^\]]*)\])/.exec(line);
        if(!m)return;
        const lo=parseInt(m[1],16),hi=parseInt(m[2],16);
        if(m[3]!==undefined){
          let cur=parseInt(m[3].replace(/\s/g,'').slice(0,4),16);
          for(let c=lo;c<=hi;c++)map.set(c,String.fromCharCode(cur++));
        }else if(m[4]!==undefined){
          const items=m[4].match(/<([0-9A-Fa-f\s]+)>/g)||[];let idx=0;
          for(let c=lo;c<=hi&&idx<items.length;c++,idx++)map.set(c,hexToUnicode(items[idx].replace(/[<>]/g,'')));
        }
      });
    });
  }
  // 解析 PDF 字面字符串转义 -> 字节数组
  function parseLiteral(str){
    const out=[];for(let i=0;i<str.length;i++){
      let c=str[i];
      if(c==='\\'){
        const n=str[i+1];
        if(n>='0'&&n<='7'){let oct='';let j=i+1;while(j<str.length&&str[j]>='0'&&str[j]<='7'&&oct.length<3){oct+=str[j];j++;}out.push(parseInt(oct,8)&0xff);i=j-1;continue;}
        const esc={n:'\n',r:'\r',t:'\t',b:'\b',f:'\f','(':'(',')':')','\\':'\\'};
        out.push((esc[n]||n).charCodeAt(0)&0xff);i++;continue;
      }
      out.push(str.charCodeAt(i)&0xff);
    }
    return out;
  }
  // 字节序列按 CMap 解码（尝试 2 字节 CID，再 1 字节）
  function decodeBytes(bytes,map){
    let out='';
    if(bytes.length>=2){
      let two='';let ok=true;const tmp=[];
      for(let i=0;i+1<bytes.length;i+=2){const cid=(bytes[i]<<8)|bytes[i+1];if(map.has(cid))tmp.push(map.get(cid));else{ok=false;break;}}
      if(ok&&tmp.length){return tmp.join('');}
    }
    for(let i=0;i<bytes.length;i++){
      if(map.has(bytes[i]))out+=map.get(bytes[i]);
      else out+=String.fromCharCode(bytes[i]);
    }
    return out;
  }
  function decodeHex(hex,map){
    hex=hex.replace(/\s/g,'');let out='';
    if(hex.length%4===0){
      let ok=true;const tmp=[];
      for(let i=0;i+4<=hex.length;i+=4){const cid=parseInt(hex.slice(i,i+4),16);if(map.has(cid))tmp.push(map.get(cid));else{ok=false;break;}}
      if(ok&&tmp.length)return tmp.join('');
    }
    for(let i=0;i+2<=hex.length;i+=2){const cid=parseInt(hex.slice(i,i+2),16);out+=map.has(cid)?map.get(cid):String.fromCharCode(cid);}
    return out;
  }
  // 从内容流抽取文本
  function extractFromContent(text,map){
    let out='';
    const blocks=text.match(/BT[\s\S]*?ET/g)||[];
    blocks.forEach(blk=>{
      let line='';
      let lastY=null;
      // 词法扫描
      let i=0;
      const pushSpace=()=>{if(line&&!/\s$/.test(line))line+=' ';};
      while(i<blk.length){
        const c=blk[i];
        // Tm 定位：a b c d e f Tm，f 为 y
        let mt=/(-?\d+\.?\d*)\s+(-?\d+\.?\d*)\s+(-?\d+\.?\d*)\s+(-?\d+\.?\d*)\s+(-?\d+\.?\d*)\s+(-?\d+\.?\d*)\s+Tm/.exec(blk.slice(i,i+160));
        if(mt&&mt.index===0){const y=parseFloat(mt[6]);if(lastY!==null&&Math.abs(y-lastY)>2){out+=(out&&!/\n$/.test(out)?'\n':'')+line;line='';}lastY=y;i+=mt[0].length;continue;}
        if(/^T\*\s*/.test(blk.slice(i,3))){out+=(line?'\n':'');line='';i+=2;continue;}
        let md=/^(-?\d+\.?\d*)\s+(-?\d+\.?\d*)\s+T[dD]/.exec(blk.slice(i,80));
        if(md&&md.index===0){const y=parseFloat(md[2]);if(Math.abs(y)>2){out+=(line?'\n':'');line='';}i+=md[0].length;continue;}
        // TJ 数组
        if(c==='['){
          const close=blk.indexOf(']',i);
          if(close>0){
            const arr=blk.slice(i+1,close);const tjM=/\]\s*TJ/.exec(blk.slice(close,close+6));
            const parts=arr.match(/<[^>]*>|\([^)]*\)|-?\d+\.?\d*/g)||[];
            parts.forEach(p=>{
              if(p[0]==='<')line+=decodeHex(p.slice(1,-1),map);
              else if(p[0]==='(')line+=decodeBytes(parseLiteral(p.slice(1,-1)),map);
              else{const n=parseFloat(p);if(n<=-80)pushSpace();}
            });
            i=close+1+(tjM?tjM[0].length:0);continue;
          }
        }
        // Tj 单串
        if(c==='('){
          let j=i+1,depth=1;
          while(j<blk.length&&depth){if(blk[j]==='\\'){j+=2;continue;}if(blk[j]==='(')depth++;else if(blk[j]===')')depth--;j++;}
          const lit=blk.slice(i+1,j-1);
          const tail=blk.slice(j,j+4);
          line+=decodeBytes(parseLiteral(lit),map);
          i=j;continue;
        }
        if(c==='<'){
          const j=blk.indexOf('>',i);
          if(j>0){line+=decodeHex(blk.slice(i+1,j),map);i=j+1;continue;}
        }
        i++;
      }
      if(line)out+=line+'\n';
    });
    return out;
  }
  async function extract(arrayBuffer){
    const bytes=new Uint8Array(arrayBuffer);
    // 第一遍：解压所有流
    const decoded=[];const cmap=new Map();
    for(const {text} of rawStreams(bytes)){
      const raw=latinBytes(text);
      let dec=await inflate(raw); // FlateDecode(zlib)；非压缩数据会返回 null
      if(dec){
        const t=new TextDecoder('utf-8',{fatal:false}).decode(dec);
        decoded.push(t);
        if(/begincmap|beginbfchar|beginbfrange/.test(t))parseCMap(t,cmap);
      }else{
        // 未压缩流：CMap 或明文内容流同样纳入
        if(/begincmap|beginbfchar|beginbfrange/.test(text)){decoded.push(text);parseCMap(text,cmap);}
        else if(/\bBT\b/.test(text))decoded.push(text);
      }
    }
    // 未压缩对象流里也可能直接含 CMap
    const latinAll=bytesToStr(bytes);
    if(/begincmap/.test(latinAll))parseCMap(latinAll,cmap);
    // 第二遍：从解压后的内容流抽文本
    let full='';
    decoded.forEach(t=>{if(/\bBT\b/.test(t))full+='\n'+extractFromContent(t,cmap);});
    // 清理：合并过度断行、去零宽、统一空白
    full=full.replace(/\x00/g,'').replace(/[ \t]+\n/g,'\n').replace(/\n{3,}/g,'\n\n').replace(/[ \t]{2,}/g,' ').trim();
    return {text:full,cmapSize:cmap.size};
  }
  return {extract};
})();
