// 中文 2-4 字词频统计（基于 N-gram 粗切，配合停用词与词长过滤）
function keywordFreq(text,topN=12){
  const clean=(text||'').replace(/[^\u4e00-\u9fa5A-Za-z0-9]/g,' ');
  const freq={};
  // 2~4 字滑窗
  for(const seg of clean.split(/\s+/)){
    for(let len=2;len<=4;len++){
      for(let i=0;i+len<=seg.length;i++){
        const w=seg.slice(i,i+len);
        if(/^[0-9]+$/.test(w))continue;
        if(STOP_WORDS.has(w))continue;
        // 首尾是停用单字的降权/跳过
        const head=w[0],tail=w[w.length-1];
        if('的了在是和与及或将把被对为以于中我你他她它们这那就都很也还而且并但由从向到给让使不没无有要会能可之其此该本各每等种时'.includes(head))continue;
        if('的了在是和与及或将把被对为以于中我你他她它们这那就都很也还而且并但由从向到给让使不没无有要会能可之其此该本各每等种时地得着过'.includes(tail))continue;
        freq[w]=(freq[w]||0)+1;
      }
    }
  }
  // 长词包含短词且频次相同则抑制短词
  const entries=Object.entries(freq).filter(([w,c])=>c>=1);
  const score={};
  entries.forEach(([w,c])=>{score[w]=c*(w.length>=3?1.6:1);});
  return entries.map(([w])=>[w,score[w]]).sort((a,b)=>b[1]-a[1]||b[0].length-a[0].length).slice(0,topN).map(x=>x[0]);
}
// 抽取式摘要：句子打分（位置、关键词、线索词、长度）
function extractSummary(text,maxSent=5){
  const sents=splitSentences(text);
  if(sents.length<=maxSent)return sents;
  const kws=new Set(keywordFreq(text,20));
  const clue=/一是|二是|三是|首先|其次|再次|最后|总体|综上|因此|总之|主要|关键|核心|重点|目标|任务|要求|原则|措施|成效|结果|表明|显示|认为|指出|强调/;
  const scored=sents.map((s,i)=>{
    let sc=0;
    for(const w of kws){if(s.includes(w))sc+=1.2;}
    if(clue.test(s))sc+=2;
    sc+=Math.max(0,2-i*0.4); // 开头句权重
    const len=s.length; if(len<10)sc-=2; if(len>120)sc-=1;
    return {s,sc,i};
  }).sort((a,b)=>b.sc-a.sc);
  return scored.slice(0,maxSent).sort((a,b)=>a.i-b.i).map(x=>x.s);
}

function buildOutline(text){
  const heads=[];
  const lines=(text||'').split(/\n/).map(x=>x.trim()).filter(Boolean);
  const reNum=/^(?:[一二三四五六七八九十]+、|（[一二三四五六七八九十]+）|\([一二三四五六七八九十]+\)|\d+[.、]|[（(]\d+[)）]|第[一二三四五六七八九十\d]+[章节部分])/;
  lines.forEach(l=>{if(reNum.test(l)&&l.length<=40)heads.push(l.replace(/[。：:\s]+$/,''));});
  if(heads.length>=2)return [...new Set(heads)].slice(0,12);
  // 无编号：用要点句生成
  return extractSummary(text,6).map((s,i)=>`${'一二三四五六七八九十'[i]}、${s.length>26?s.slice(0,26)+'…':s}`);
}
