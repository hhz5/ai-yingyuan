      const mine=[...S.myAgents].sort((a,b)=>b.ut-a.ut);
      const others=MARKET.filter(m=>!S.myAgents.some(x=>x.name===m.name));
      list=[...mine.map(a=>({a,mine:true})),...others.map(a=>({a,mine:false}))];

      const minePub=S.myAgents.filter(a=>a.status==='published'&&a.visibility==='public'&&a.category===curTab).map(a=>({a,mine:true}));

  }
  const discoverSec=h('section',{},[
    h('div',{class:'section-head'},[h('h2',{text:'发现更多'})]),

  ]);
  // ---- 公文范文（展示前 4 篇，全部进入范文库） ----
  const tplSec=h('section',{},[
    h('div',{class:'section-head'},[h('h2',{text:'公文范文'}),h('button',{class:'btn btn-sm',html:ICON.arrowR+' 进入范文库',onClick:()=>go('#/templates')})]),

    ])))
  ]);
  view.appendChild(h('div',{class:'page'},[recoSec,discoverSec,tplSec]));
  renderTab();
};

  const g=ctx.createLinearGradient(0,0,120,120);g.addColorStop(0,colors[0]);g.addColorStop(1,colors[1]);
  ctx.fillStyle=g;ctx.beginPath();ctx.roundRect(0,0,120,120,26);ctx.fill();
  ctx.fillStyle='#fff';ctx.font='700 58px "Noto Sans SC",sans-serif';ctx.textAlign='center';ctx.textBaseline='middle';ctx.fillText(ch,60,66);
  return cv.toDataURL('image/png');
}

function openOptimize(agent, personaTa){
  if(!agent.persona.trim()){toast('请先填写人设与回复逻辑','warn');return;}

  const retry=h('button',{class:'btn btn-sm',html:ICON.refresh+' 重试'});
  let ctrl=null;
  function run(){
    useBtn.disabled=true;useBtn.classList.add('btn-disabled');ta.value='';
    const full=AI.optimize(agent.persona);
    const opSc=makeAutoScroll(ta);
    ctrl=streamText(full,(t,done)=>{ta.value=t;opSc(done);},{} );
    ctrl.start().then(()=>{useBtn.disabled=false;useBtn.classList.remove('btn-disabled');});
  }
  retry.onclick=run;
  useBtn.onclick=()=>{agent.persona=ta.value;personaTa.value=ta.value;saveState();closeModal();toast('已应用优化后的提示词');};
  openModal({width:680,title:'提示词优化',body:[
    h('div',{style:{display:'flex',justifyContent:'flex-end',marginBottom:'8px'}},[retry]),ta
  ],footer:[h('button',{class:'btn',text:'取消',onClick:closeModal}),useBtn]});
  run();
}
